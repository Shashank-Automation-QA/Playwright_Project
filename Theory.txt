// =====================================================
// ARROW FUNCTIONS IN TYPESCRIPT - QUICK NOTES
// =====================================================

// 1. No Parameter Arrow Function

const showMessage = () => {
    console.log("Hello");
};

// Syntax:
// () => {}


// =====================================================

// 2. Single Parameter Arrow Function

const greet = (name: string) => {
    console.log(name);
};

// Syntax:
// (parameter) => {}


// =====================================================

// 3. Multiple Parameter Arrow Function

const add = (a: number, b: number) => {
    return a + b;
};

// Syntax:
// (parameter1, parameter2) => {}


// =====================================================

// 4. Implicit Return Arrow Function

const square = (num: number) => num * num;

// Syntax:
// (parameter) => expression

// Equivalent to:

const square2 = (num: number) => {
    return num * num;
};


// =====================================================

// 5. Explicit Return Arrow Function

const multiply = (a: number, b: number) => {
    return a * b;
};

// Syntax:
// (parameter) => {
//     return value;
// }


// =====================================================

// 6. Arrow Function with Return Type

const getMessage = (name: string): string => {
    return `Hello ${name}`;
};

// Syntax:
// (parameter): returnType => {}


// =====================================================

// 7. Async Arrow Function

const getData = async () => {
    console.log("Fetching Data");
};

// Syntax:
// async () => {}


// =====================================================

// 8. Async Arrow Function with Parameter

const getUser = async (id: number) => {
    console.log(id);
};

// Syntax:
// async (parameter) => {}


// =====================================================

// 9. Async Arrow Function with Return Type

const getResult = async (): Promise<string> => {
    return "PASS";
};

// Syntax:
// async (): Promise<Type> => {}


// =====================================================

// 10. Object Destructuring Arrow Function

const displayUser = ({ name, age }) => {
    console.log(name);
    console.log(age);
};

// Syntax:
// ({ property1, property2 }) => {}


// =====================================================

// 11. Array map() Arrow Function

const ids = ["TC-001", "TC-002"];

const updatedIds = ids.map(id => `Run-${id}`);

// Syntax:
// item => expression


// =====================================================

// 12. Array filter() Arrow Function

const testCases = ["TC-001", "TC-002", "TC-003"];

const regressionCases = testCases.filter(
    tc => tc !== "TC-003"
);

// Syntax:
// item => condition


// =====================================================

// 13. Array forEach() Arrow Function

const users = ["Shashank", "Rahul", "Amit"];

users.forEach(user => {
    console.log(user);
});

// Syntax:
// item => {}


// =====================================================

// 14. Playwright Test Arrow Function

test('has title', async ({ page }) => {

});

// Syntax:
// async ({ page }) => {}


// =====================================================

// 15. Playwright describe Arrow Function

test.describe('Smoke Tests', () => {

});

// Syntax:
// () => {}


// =====================================================

// MOST COMMON INTERVIEW SYNTAXES
// =====================================================

// No Parameter
() => {}

// One Parameter
(name) => {}

// Multiple Parameters
(a, b) => {}

// Implicit Return
name => name.toUpperCase()

// Explicit Return
(name) => {
    return name.toUpperCase();
}

// Async Function
async () => {}

// Async with Parameter
async (id) => {}

// Return Type
(name): string => {}

// Object Destructuring
({ name, age }) => {}

// Playwright Style
async ({ page }) => {}
